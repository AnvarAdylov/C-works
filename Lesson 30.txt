int partition(student *x[], int first, int last) {
	int pivot = first;

	int up = first, down = last;
	while (up < down) {

		while (compare1(x[up], x[pivot]) && up < last) {
			up++;
		}
		while (compare2(x[down], x[pivot])) {
			down--;
		}
		if (up < down) {
#ifdef USE_NAMED_STUDENT
			student temp;
			temp.id = x[up]->id;
			strcpy(temp.name, x[up]->name);
			x[up]->id = x[down]->id;
			strcpy(x[up]->name, x[down]->name);
			x[down]->id = temp.id;
			strcpy(x[down]->name, temp.name);
#endif
#ifdef USE_YEAR_MAJOR_STUDENT
			student temp;
			temp.id = x[up]->id;
			temp.year = x[up]->year;
			strcpy(temp.major, x[up]->major);

			x[up]->id = x[down]->id;
			x[up]->year = x[down]->year;
			strcpy(x[up]->major, x[down]->major);

			x[down]->id = temp.id;
			x[down]->year = temp.year;
			strcpy(x[down]->major, temp.major);
#endif
		}
	}
#ifdef USE_NAMED_STUDENT
	student temp2;
	temp2.id = x[pivot]->id;
	strcpy(temp2.name, x[pivot]->name);
	x[pivot]->id = x[down]->id;
	strcpy(x[pivot]->name, x[down]->name);
	x[down]->id = temp2.id;
	strcpy(x[down]->name, temp2.name);
#endif
#ifdef USE_YEAR_MAJOR_STUDENT
	student temp2;
	temp2.id = x[pivot]->id;
	temp2.year = x[pivot]->year;
	strcpy(temp2.major, x[pivot]->major);

	x[pivot]->id = x[down]->id;
	x[pivot]->year = x[down]->year;
	strcpy(x[pivot]->major, x[down]->major);

	x[down]->id = temp2.id;
	x[down]->year = temp2.year;
	strcpy(x[down]->major, temp2.major);
#endif
	return down;
	// Put your quicksort partition code here
}

void quicksort(student *x[], int first, int last) {
	if (first < last) {
		int pivIndex = partition(x, first, last);
		quicksort(x, first, pivIndex - 1);
		quicksort(x, pivIndex + 1, last);
	}
}
void sort(student *x[]) {
	quicksort(x, 0, SIZE - 1);
}
