// function to estimate size of the string
int sizeofstring(char *string) {
	int n = 0;
	while (*string != '\0') {
		n = n + 1;
		string++;
	}
	return n;
}

	int i = sizeofstring(source);
	int j;
char *target = malloc(sizeof(char)*(i+1));
	for (j = 0; j<i; j++){
		*target = *(source + j);
		target++;
		}
*target = '\0'; 

target = target - i;
return target;
}

_Bool nameCompare(char name1[], char name2[]) {
	int i = 0;
	do {
		if (name1[i] < name2[i]) {
			return 1;
		}
		else if (name1[i] > name2[i]) {
			return 0;
		}
		i++;
	} while (name1[i] != '\0' || name2[i] != '\0');
	return 0;
}

// Insertion sort algorithm
void insertionSort(person_count *x[], int size) {
	int i;
	for (i = 0; i < size; i++) {
		int toInsertm = x[i]->males;
		int toInsertf = x[i]->females;
		char *toInserts;
		toInserts = copy_string(x[i]->country);
		int j;
		for (j = i; j >= 0; j--) {
			if (j == 0 || x[j - 1]->males >= toInsertm) {

				x[j]->males = toInsertm;
				x[j]->females = toInsertf;
				x[j]->country = copy_string(toInserts);
				break;
			}
			else {
				x[j]->males = x[j - 1]->males;
				x[j]->females = x[j - 1]->females;
				x[j]->country = copy_string(x[j - 1]->country);
			}
		}
	}
}

// Partition function
int partition(person_count *x[], int first, int last) {

	int pivot = first;

	int up = first, down = last;
	while (up < down) {

		while (totalEnroll(x[up]) >= totalEnroll(x[pivot]) && up < last) {
			up++;
		}
		while (totalEnroll(x[down]) < totalEnroll(x[pivot])) {
			down--;
		}
		if (up < down) {
			int tempf = x[up]->females;
			int tempm = x[up]->males;
			char *temps;
			temps = copy_string(x[up]->country);
			x[up]->females = x[down]->females;
			x[up]->males = x[down]->males;
			x[up]->country = copy_string(x[down]->country);
			x[down]->females = tempf;
			x[down]->males = tempm;
			x[down]->country = copy_string(temps);
		}
	}

	int tempf2 = x[pivot]->females;
	int tempm2 = x[pivot]->males;
	char *temps2;
	temps2 = copy_string(x[pivot]->country);
	x[pivot]->females = x[down]->females;
	x[pivot]->males = x[down]->males;
	x[pivot]->country = copy_string(x[down]->country);
	x[down]->females = tempf2;
	x[down]->males = tempm2;
	x[down]->country = copy_string(temps2);

	return down;
}
// Quicksort algorithm
void quicksort(person_count *x[], int first, int last) {

	if (first < last) {
		int pivIndex = partition(x, first, last);
		quicksort(x, first, pivIndex - 1);
		quicksort(x, pivIndex + 1, last);
	}
}
