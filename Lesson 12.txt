int main()
{
	int e;
	printf("To exit the programm input the negative value: \n");
	scanf("%i", &e);
	while (e > 0) {
		int factorial[10], i = 0;
		do
		{
			factorial[i] = factorial[i - 1] * i;
			if (factorial[i] == 0) {
				factorial[i] = 1;
			}
			printf("%i \n", factorial[i]);
			i = i + 1;
		} while (i < 10);
		printf("To exit the programm input the negative value: \n");
		scanf("%i", &e);
	}
	return 0;
}

#include <stdlib.h>


int main()
{
	int e;
	printf("To exit the programm input the negative value: \n");
	scanf("%i", &e);
	while (e > 0) {
		int random_numbers[20], i = 0, max = 0, min = 100, x, sum = 0, n = 0;
		double avg;
		do {
			x = rand() % 51;
			random_numbers[i] = x + 50;
			i = i + 1;
		} while (i < 20);
		for (i = 0; i < 20; i++)
		{
			printf("%i \n", random_numbers[i]);
			if (max <= random_numbers[i]) {
				max = random_numbers[i];
			}
			if (min >= random_numbers[i]) {
				min = random_numbers[i];
			}
			sum = sum + random_numbers[i];
			n = n + 1;
		}
		avg = sum/n;
		printf("The max value in array is %i \n", max);
		printf("The min value in array is %i \n", min);
		printf("The average for this array is %f \n", avg);
		printf("To exit the programm input the negative value: \n");
		scanf("%i", &e);
	}
	return 0;
}
