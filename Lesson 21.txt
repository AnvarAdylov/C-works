#include <string.h>
#include <math.h>
double Power(double x, int n) {
	if (n == 0) {
		return 1.00;
	}
	else if (n>0) {
		return Power(x, n - 1)*x;
	}
	else if (n<0) {
		return 1 / (Power(x, (-n) - 1)*x);
	}
}
