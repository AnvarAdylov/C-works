#include <stdio.h>

typedef struct {
	int id;
	char name[12];
} student;
void copy_string(char *source, char *target) {
	while (*source) {
		*target = *source;
		source++;
		target++;
	}
	*target = '\0';
}

_Bool nameCompare(char name1[], char name2[]) {
	int i = 0;
	do {
		if (name1[i] < name2[i]) {
			return 1;
		}
		else if (name1[i] > name2[i]) {
			return 0;
		}
		i++;
	} while (name1[i] != '\0' || name2[i] != '\0');
	return 0;
}
void insertionSortByName(student *x[], int size) {

	int i;
	for (i = 0; i < size; i++) {
		student toInsert;
		toInsert.id = x[i]->id;
		copy_string(x[i]->name, toInsert.name);

		int j;
		for (j = i; j >= 0; j--) {
						
			if (j == 0) {
				x[j]->id = toInsert.id;
				copy_string(toInsert.name, x[j]->name);
				break;

			}
			else if (nameCompare(x[j-1]->name, toInsert.name) == 1) {
				x[j]->id = toInsert.id;
				copy_string(toInsert.name, x[j]->name);
				break;
			} else {
				x[j]->id = x[j - 1]->id;
				copy_string(x[j - 1]->name, x[j]->name);
			}
		}
	}
}
void printArray(student *x[], int size) {

	int i;
	for (i = 0; i < size; i++) {
		printf("%8i %12s \n", x[i]->id, x[i]->name);
	}
	printf("\n");
}
