typedef struct {
	int CustomerID;
	char type;
	int accbalance;
} BankCustomer;

int main()
{
	int i, sumc = 0, sums = 0, sumd = 0, nc = 0, ns = 0, nd = 0;
	double avgc, avgs, avgd;
	FILE *infile;
	infile = fopen("clients.txt", "r");
	BankCustomer list[10];
	for (i = 0; i < 10; i++) {
		fscanf(infile, "%i %c %i", &list[i].CustomerID, &list[i].type, &list[i].accbalance);
	}
	for (i = 0; i < 10; i++) {
		if (list[i].type == 'c' || list[i].type == 'C') {
			sumc = sumc + list[i].accbalance;
			nc = nc + 1;
		}

		if (list[i].type == 's' || list[i].type == 'S') {
			sums = sums + list[i].accbalance;
			ns = ns + 1;
		}

		if (list[i].type == 'd' || list[i].type == 'D') {
			sumd = sumd + list[i].accbalance;
			nd = nd + 1;
		}
	}
	avgc = sumc / nc;
	avgs = sums / ns;
	avgd = sumd / nd;
	printf("The average balance on checking accounts %f \n", avgc);
	printf("The average balance on savings accounts %f \n", avgs);
	printf("The average balance on deposit accounts %f \n", avgd);
	return 0;
}
