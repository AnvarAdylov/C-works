#include <stdio.h>
void reverse(char *fromStr, char *toStr) {
	char init;
	init = *fromStr;
	while (*fromStr != '\0') {
		fromStr++;
	}
	while (*fromStr != init) {
		fromStr--;
	*toStr = *fromStr;
	toStr++;
	} 
	*toStr = '\0';
}
int main() {
	char a[] = {"Whatever"}, b[20];
	reverse(a, b);
	printf("%s", b);
}

#include <stdio.h>
void concat(char *str1, char *str2, char *resultStr) {
	do {
		*resultStr = *str1;
		str1++;
		resultStr++;
	} while (*str1 != '\0');
	do {
		*resultStr = *str2;
		str2++;
		resultStr++;
	} while (*str2 != '\0');
	*resultStr = '\0';
}
int main() {
	char a[] = {"Hello "}, b[] = {"me!"}, c[30];
	concat(a, b, c);
	printf("%s", c);
}
