int main()
{
	int y = 3, x = 4;
	int a[y][x], b[x][y];

	int i, j, val = 100;

	/* set matrix elements */
	for (i = 0; i < y; i++) {
		for (j = 0; j < x; j++) {
			a[i][j] = val;
			val++;
		}
	}
	for (i = 0; i < x; i++) {
		for (j = 0; j < y; j++) {
			b[i][j] = a[j][i] + 25;
		}
	}
	
	for (i = 0; i < x; i++) {
		for (j = 0; j < y; j++) {
			printf("%5d", b[i][j]);
		}
		printf("\n");
	}
	return 0;
}

#include <stdlib.h>


int main()
{

	int e;
	printf("To exit the programm input the negative value: \n");
	scanf("%i", &e);
	while (e > 0) {
		int y = 5, x = 5, max = 0, imax, jmax;
		int a[y][x];

		int i, j;

		for (i = 0; i < y; i++) {
			for (j = 0; j < x; j++) {
				a[i][j] = rand() % 10;
			}
		}

		for (i = 0; i < y; i++) {
			for (j = 0; j < x; j++) {
				if (a[i][j]>max) {
					max = a[i][j];
					imax = i;
					jmax = j;
				}
				printf("%5d ", a[i][j]);
			}
			printf("\n");
		}
		printf("the maximum value is %i and it has coordinates of y = %i and x = %i \n", max, imax++, jmax++);
		printf("To exit the programm input the negative value: \n");
		scanf("%i", &e);
	}
	return 0;
}

#include <stdlib.h>


int main()
{
	int e;
	printf("To exit the programm input the negative value: \n");
	scanf("%i", &e);
	while (e > 0) {
		int x = 3, y = 3;
		int a[3][3], b[3][3], c[3][3];
		int i, j, k;
		for (i = 0; i < y; i++) {
			for (j = 0; j < x; j++) {
				a[i][j] = rand() % 10;
				b[i][j] = rand() % 10;
			}
		}

		for (i = 0; i < y; i++) {
			for (j = 0; j < x; j++) {
				printf("%5d", a[i][j]);
			}
			printf("\n");
		}
		for (i = 0; i < y; i++) {
			for (j = 0; j < x; j++) {
				printf("%5d", b[i][j]);
			}
			printf("\n");
		}
		for (i = 0; i < y; i++) {
			for (j = 0; j < x; j++) {
				c[i][j] = 0;
			}
		}
		for (i = 0; i < y; i++) {

			for (j = 0; j < x; j++) {
				for (k = 0; k < y; k++) {
					c[i][j] = c[i][j] + a[i][k] * b[k][j];
				}
			}
			printf("\n");
		}
		for (i = 0; i < y; i++) {
			for (j = 0; j < x; j++) {
				printf("%5d", c[i][j]);
			}
			printf("\n");
		}

		printf("To exit the programm input the negative value: \n");
		scanf("%i", &e);
	}
	return 0;
}
